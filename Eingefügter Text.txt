<script>
  const state = {
    uber: { online: true, trips: 0 },
    bolt: { online: true, trips: 0 },
    activeTrip: null,
    tripSeconds: 0,
    tripInterval: null,
    toastTimeout: null,
    logCount: 0
  };

  function now() {
    const d = new Date();
    return d.getHours().toString().padStart(2,'0') + ':' +
           d.getMinutes().toString().padStart(2,'0') + ':' +
           d.getSeconds().toString().padStart(2,'0');
  }

  function addLog(icon, iconClass, message, subtext, tagHtml) {
    state.logCount++;
    const container = document.getElementById('log-container');
    const entry = document.createElement('div');
    entry.className = 'log-entry';
    entry.innerHTML = `
      <div class="log-icon ${iconClass}">${icon}</div>
      <div class="log-text">
        <div class="log-message">${message} ${tagHtml || ''}</div>
        <div class="log-time">${now()} — ${subtext}</div>
      </div>`;
    container.insertBefore(entry, container.firstChild);
    if (state.logCount > 8) {
      container.removeChild(container.lastChild);
    }
  }

  function showToast(title, sub) {
    clearTimeout(state.toastTimeout);
    document.getElementById('toast-title').textContent = title;
    document.getElementById('toast-sub').textContent = sub;
    const t = document.getElementById('toast');
    t.classList.add('show');
    state.toastTimeout = setTimeout(() => t.classList.remove('show'), 3000);
  }

  function updatePlatformUI(platform, online) {
    const card = document.getElementById(platform + '-card');
    const toggle = document.getElementById(platform + '-toggle');
    const badge = document.getElementById(platform + '-badge');
    const statusText = document.getElementById(platform + '-status-text');

    if (online) {
      card.className = 'platform-card online-' + platform;
      toggle.className = 'platform-toggle on-' + platform;
      badge.className = 'platform-status-badge badge-online-' + platform;
      statusText.textContent = 'Online';
    } else {
      card.className = 'platform-card offline-state';
      toggle.className = 'platform-toggle';
      badge.className = 'platform-status-badge badge-offline';
      statusText.textContent = 'Offline';
    }
  }

  function setOnline(platform, online, reason) {
    state[platform].online = online;
    updatePlatformUI(platform, online);
  }

  function simulateUberTrip() {
    if (state.activeTrip) { showToast('Bereits aktiv', 'Bitte zuerst Fahrt beenden'); return; }

    addLog('🚗', 'info', 'Uber: Auftrag erkannt', 'Fahrer hat Fahrt angenommen', '<span class="log-platform-tag tag-uber">UBER</span>');

    setTimeout(() => {
      setOnline('bolt', false);
      addLog('⚡', 'warning', 'Bolt → Offline gestellt', 'Automatisch deaktiviert', '<span class="log-platform-tag tag-bolt">BOLT</span>');
      addLog('✅', 'success', 'Schutz aktiv — Bolt offline', 'Keine weiteren Bolt-Aufträge möglich', '');
      showToast('Bolt automatisch offline', 'Uber-Fahrt läuft — kein doppelter Auftrag');
      startTrip('UBER');
    }, 800);

    state.uber.trips++;
    document.getElementById('uber-trips').textContent = state.uber.trips + ' Fahrten';
  }

  function simulateBoltTrip() {
    if (state.activeTrip) { showToast('Bereits aktiv', 'Bitte zuerst Fahrt beenden'); return; }

    addLog('⚡', 'info', 'Bolt: Auftrag erkannt', 'Fahrer hat Fahrt angenommen', '<span class="log-platform-tag tag-bolt">BOLT</span>');

    setTimeout(() => {
      setOnline('uber', false);
      addLog('🚗', 'warning', 'Uber → Offline gestellt', 'Automatisch deaktiviert', '<span class="log-platform-tag tag-uber">UBER</span>');
      addLog('✅', 'success', 'Schutz aktiv — Uber offline', 'Keine weiteren Uber-Aufträge möglich', '');
      showToast('Uber automatisch offline', 'Bolt-Fahrt läuft — kein doppelter Auftrag');
      startTrip('BOLT');
    }, 800);

    state.bolt.trips++;
    document.getElementById('bolt-trips').textContent = state.bolt.trips + ' Fahrten';
  }

  function startTrip(platform) {
    state.activeTrip = platform;
    state.tripSeconds = 0;
    document.getElementById('trip-card').classList.add('visible');

    clearInterval(state.tripInterval);
    state.tripInterval = setInterval(() => {
      state.tripSeconds++;
      const m = Math.floor(state.tripSeconds / 60).toString().padStart(2,'0');
      const s = (state.tripSeconds % 60).toString().padStart(2,'0');
      document.getElementById('trip-timer').textContent = m + ':' + s;
    }, 1000);
  }

  function simulateTripEnd() {
    if (!state.activeTrip) { showToast('Keine aktive Fahrt', 'Starte zuerst eine Demo-Fahrt'); return; }

    const platform = state.activeTrip;
    clearInterval(state.tripInterval);
    state.activeTrip = null;

    document.getElementById('trip-card').classList.remove('visible');
    document.getElementById('trip-timer').textContent = '00:00';

    addLog('🏁', 'success', 'Fahrt beendet', 'Plattformen wieder verfügbar', '');

    setTimeout(() => {
      setOnline('uber', true);
      setOnline('bolt', true);
      addLog('✅', 'success', 'Beide Plattformen online', 'Bereit für neue Aufträge', '');
      showToast('Fahrt beendet', 'Uber & Bolt sind wieder online');
    }, 600);
  }

  function allOnline() {
    if (state.activeTrip) { showToast('Fahrt aktiv', 'Fahrt zuerst beenden'); return; }
    setOnline('uber', true);
    setOnline('bolt', true);
    addLog('✅', 'success', 'Alle Plattformen online', 'Manuell aktiviert', '');
    showToast('Alle online', 'Uber & Bolt aktiv');
  }

  function allOffline() {
    setOnline('uber', false);
    setOnline('bolt', false);
    addLog('⏸', 'warning', 'Alle Plattformen offline', 'Manuell deaktiviert', '');
    showToast('Alle offline', 'Pause aktiv — keine Aufträge');
  }

  function manualToggle(platform) {
    const newState = !state[platform].online;
    setOnline(platform, newState);
    const label = platform === 'uber' ? 'Uber' : 'Bolt';
    const status = newState ? 'Online' : 'Offline';
    addLog(newState ? '✅' : '⏸', newState ? 'success' : 'warning',
      `${label} → ${status}`, 'Manuell umgeschaltet',
      `<span class="log-platform-tag tag-${platform}">${platform.toUpperCase()}</span>`);
  }

  function clearLog() {
    const container = document.getElementById('log-container');
    container.innerHTML = `
      <div class="log-entry">
        <div class="log-icon info">🛡️</div>
        <div class="log-text">
          <div class="log-message">Log geleert <span class="log-platform-tag tag-uber">SYSTEM</span></div>
          <div class="log-time">${now()} — Bereit</div>
        </div>
      </div>`;
    state.logCount = 1;
  }
</script>
